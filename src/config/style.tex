% --- Scelta del Tema Core ---
\usetheme{metropolis}
% Rimosso owl per evitare conflitti di contrasto sullo sfondo bianco.

% --- Palette Colori Avanzata (Corporate Industrial) ---
\definecolor{energyDark}{RGB}{11, 29, 48}     % Blu Notte Profondo (Massimo contrasto per i testi)
\definecolor{energyTeal}{RGB}{49, 151, 149}   % Verde Smeraldo / Brand Color per accenti strategici
\definecolor{energyLight}{RGB}{245, 247, 250} % Grigio Liquido (Sfondo ultra-morbido per i blocchi)
\definecolor{alertRed}{RGB}{197, 48, 48}      % Rosso Desaturato per i blocchi alert (elegante, non aggressivo)
\definecolor{textMuted}{RGB}{74, 85, 104}     % Grigio Fumo per descrizioni secondarie e liste

% --- Assegnazione Colori Strutturali ---
\setbeamercolor{background canvas}{bg=white}
\setbeamercolor{normal text}{fg=energyDark}
\setbeamercolor{alerted text}{fg=energyTeal}
\setbeamercolor{example text}{fg=energyTeal}

% --- Configurazione Barra dei Titoli (Header) ---
\setbeamercolor{frametitle}{bg=energyDark, fg=white}
\setbeamercolor{titleline}{bg=energyTeal} % Sottile linea smeraldo sotto l'header

% --- Strutturazione Liste ed Elenchi Puntati ---
\setbeamercolor{structure}{fg=energyTeal} % Forza proiettili ed elementi grafici in Teal
\setbeamercolor{item}{fg=energyTeal}
\setbeamercolor{subitem}{fg=textMuted}

% --- Personalizzazione dei Blocchi di Testo (Stile Elegante "Filled") ---
\metroset{block=fill} % Cambiato da transparent a fill: crea scatole di contenuto pulite

% 1. Blocco Standard (Notizie / Dati regolati)
\setbeamercolor{block title}{bg=energyDark, fg=white}
\setbeamercolor{block body}{bg=energyLight, fg=energyDark}

% 2. Blocco Alert (Rischio Commodity / Cambi FX)
\setbeamercolor{block title alert}{bg=alertRed, fg=white}
\setbeamercolor{block body alert}{bg=energyLight, fg=energyDark}

% 3. Blocco Esempio / Focus (Innovazione / VAS)
\setbeamercolor{block title example}{bg=energyTeal, fg=white}
\setbeamercolor{block body example}{bg=energyLight, fg=energyDark}

% --- Ottimizzazioni Tipografiche e Geometrie ---
\setbeamerfont{title}{size=\Large, series=\bfseries, family=\sffamily}
\setbeamerfont{frametitle}{size=\large, series=\bfseries, family=\sffamily}
\setbeamerfont{block title}{size=\normalfont\small, series=\bfseries}
\setbeamerfont{block body}{size=\small}

% --- Rimozione Elementi di Distrazione Visiva ---
\setbeamertemplate{navigation symbols}{} % Rimuove i micro-bottoni di navigazione in basso
\setbeamercolor{page number in head/foot}{fg=textMuted} % Numero di pagina discreto
